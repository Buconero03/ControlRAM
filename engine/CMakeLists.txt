cmake_minimum_required(VERSION 3.16)
project(AdvancedMemoryEngine VERSION 0.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
add_executable(ame_resource_analysis_tests tests/resource_analysis_tests.cpp)
add_test(NAME ame_resource_analysis COMMAND ame_resource_analysis_tests)
add_executable(ame_policy_tests tests/policy_tests.cpp)
add_test(NAME ame_policy COMMAND ame_policy_tests)
if(WIN32)
  add_library(ame_resource_api SHARED src/resource_api.cpp)
  set_target_properties(ame_resource_api PROPERTIES OUTPUT_NAME ame-resource-api PREFIX "")
  target_compile_definitions(ame_resource_api PRIVATE AME_RESOURCE_API_EXPORTS _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_link_libraries(ame_resource_api PRIVATE psapi advapi32)
  add_executable(ame_resource_demo examples/resource_api_demo.cpp)
  set_target_properties(ame_resource_demo PROPERTIES OUTPUT_NAME ame-resource-demo)
  target_compile_definitions(ame_resource_demo PRIVATE _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_link_libraries(ame_resource_demo PRIVATE ame_resource_api)
  add_test(NAME ame_resource_windows COMMAND ame_resource_demo --self-test)
  set_tests_properties(ame_resource_windows PROPERTIES TIMEOUT 30)
  if(MINGW)
    target_link_options(ame_resource_demo PRIVATE -municode)
  endif()
  add_executable(ame_browser_host src/browser_host.cpp)
  set_target_properties(ame_browser_host PROPERTIES OUTPUT_NAME ame-browser-host)
  target_compile_definitions(ame_browser_host PRIVATE _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_link_libraries(ame_browser_host PRIVATE advapi32)
  add_executable(ame src/main.cpp)
  target_compile_definitions(ame PRIVATE _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_link_libraries(ame PRIVATE psapi ws2_32 user32 advapi32)
  if(MINGW)
    target_link_options(ame PRIVATE -municode)
    target_link_options(ame_browser_host PRIVATE -municode)
  endif()
endif()
foreach(resource_target ame_resource_analysis_tests ame_resource_api ame_resource_demo)
  if(TARGET ${resource_target})
    if(MSVC)
      target_compile_options(${resource_target} PRIVATE /W4 /WX /permissive- /utf-8)
    else()
      target_compile_options(${resource_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
  endif()
endforeach()
if(MSVC)
  target_compile_options(ame_policy_tests PRIVATE /W4 /permissive- /utf-8)
  if(TARGET ame)
    target_compile_options(ame PRIVATE /W4 /permissive- /utf-8)
  endif()
else()
  target_compile_options(ame_policy_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(TARGET ame)
    target_compile_options(ame PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()

if(TARGET ame_browser_host)
  if(MSVC)
    target_compile_options(ame_browser_host PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(ame_browser_host PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endif()

add_executable(ame_recover_tests tests/recover_policy_tests.cpp)
add_test(NAME ame_recover COMMAND ame_recover_tests)
if(MSVC)
  target_compile_options(ame_recover_tests PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(ame_recover_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_executable(ame_recover_replay_tests tests/recover_controller_replay_tests.cpp)
add_test(NAME ame_recover_replay COMMAND ame_recover_replay_tests)
if(MSVC)
  target_compile_options(ame_recover_replay_tests PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(ame_recover_replay_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
